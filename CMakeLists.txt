cmake_minimum_required(VERSION 3.16)
project(VoiceCraftStudio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
add_library(rnnoise STATIC vendor/rnnoise/src/denoise.c vendor/rnnoise/src/rnn.c vendor/rnnoise/src/rnn_data.c vendor/rnnoise/src/pitch.c vendor/rnnoise/src/kiss_fft.c vendor/rnnoise/src/celt_lpc.c)
target_include_directories(rnnoise PUBLIC vendor/rnnoise/include PRIVATE vendor/rnnoise/src)
target_compile_options(rnnoise PRIVATE -O2)
target_link_libraries(rnnoise PUBLIC m)
add_library(voicecraft-dsp native/dsp.cpp)
target_link_libraries(voicecraft-dsp PUBLIC rnnoise)
target_include_directories(voicecraft-dsp PUBLIC native)
add_executable(voicecraft-studio native/main.cpp native/audio.cpp native/audio.h native/window.cpp native/window.h native/timeline.cpp native/timeline.h)
target_compile_options(voicecraft-studio PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(voicecraft-studio PRIVATE voicecraft-dsp Qt5::Widgets PkgConfig::PULSE)
add_executable(voicecraft-dsp-test native/test_dsp.cpp)
target_link_libraries(voicecraft-dsp-test PRIVATE voicecraft-dsp)
add_executable(voicecraft-pcm native/pcm.cpp)
target_link_libraries(voicecraft-pcm PRIVATE voicecraft-dsp)
enable_testing()
add_test(NAME dsp COMMAND voicecraft-dsp-test)
install(TARGETS voicecraft-studio RUNTIME DESTINATION bin)
install(FILES assets/voicecraft-studio.desktop DESTINATION share/applications)
install(FILES assets/voicecraft-studio.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE vendor/rnnoise/COPYING DESTINATION share/doc/voicecraft-studio)
find_package(Qt5 5.12 REQUIRED COMPONENTS Test)
add_executable(voicecraft-gui-test native/test_gui.cpp native/window.cpp native/window.h native/timeline.cpp native/timeline.h native/audio.cpp native/audio.h)
target_link_libraries(voicecraft-gui-test PRIVATE voicecraft-dsp Qt5::Widgets Qt5::Test PkgConfig::PULSE)
install(FILES README.md VERIFICATION.md docs/USER_GUIDE.md docs/OBS_GUIDE.md docs/TROUBLESHOOTING.md docs/ARCHITECTURE.md DESTINATION share/doc/voicecraft-studio)
